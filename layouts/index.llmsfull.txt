{{- /* llms-full.txt — the content llms.txt indexes, in one request.

       Markdown rather than the rendered HTML, and .RenderShortcodes rather than
       .Plain: an answer engine wants the headings, the lists and the code
       fences, which is exactly what .Plain throws away and the markdown keeps.

       Each post is preceded by its URL, so a model quoting from this file has
       the citable address next to the text rather than having to match a title
       back to the index. That is the whole difference between a corpus and a
       source, and it is the only reason this file is worth publishing.

       Posts are separated by a `--- post: <url> ---` line. A bare `---` would
       be ambiguous with a thematic break written inside a post, which is
       ordinary markdown someone writes without thinking about this file at all.

       Every label here — `- URL:`, `- Published:`, `- Words:` — is English on
       every language's copy, deliberately. They are keys, and a reader parsing
       this file should not have to know the site's language to find them; the
       values are what carries the language, and `- Language:` states which.

       Built as a list of lines and joined, for the reason llms.txt gives: plain
       text has no forgiving renderer behind it.

       One per language, the same way llms.txt is. */ -}}
{{- $siteName := or site.Params.title site.Title -}}
{{- $siteDesc := partialCached "aeo-params-scalar.html" (dict "value" site.Params.description "path" "[params.description]") "description" site.Language.Lang -}}
{{- $aeo := partialCached "aeo-params-table.html" (dict "value" site.Params.aeo "path" "[params.aeo]") "aeo" site.Language.Lang -}}
{{- $postSections := partialCached "aeo-params-list.html" (dict "value" $aeo.postSections "path" "[params.aeo] postSections" "maps" false) "aeo.postSections" site.Language.Lang -}}
{{- if not $postSections }}{{ $postSections = slice "blogs" "posts" }}{{ end -}}
{{- $gate := partial "aeo-gate.html" . -}}
{{- $posts := where site.RegularPages "Type" "in" $postSections -}}
{{- if not $posts }}{{ $posts = site.RegularPages }}{{ end -}}
{{- $posts = partial "aeo-allowed.html" (dict "pages" $posts "disallow" $gate.disallow) -}}

{{- if not $gate.indexable -}}
# {{ $siteName }}

> This build is not for indexing: it is not a production build, and
> `allowIndexing` is not set. No post bodies are published here.
{{- else -}}
{{- $out := slice (printf "# %s" $siteName) "" -}}
{{- with $siteDesc }}{{ $out = $out | append (printf "> %s" (partial "aeo-text.html" .)) "" }}{{ end -}}
{{- $out = $out | append (printf "- Language: %s" (or site.Language.Locale site.Language.Lang)) -}}
{{- $out = $out | append (printf "- Home: %s" .Permalink) -}}
{{- $out = $out | append (printf "- Posts: %d" (len $posts)) -}}

{{- range $posts -}}
  {{- /* The boundary carries the URL rather than being a bare `---`. A post is
         markdown and markdown has thematic breaks: a `---` on a line of its own
         inside someone's post is indistinguishable from the line that separates
         two posts, and so is the setext underline under a heading. A reader
         segmenting this file had no way to tell them apart. This line cannot
         occur in prose by accident, and it answers the question the split is
         being made to answer — which post starts here. */ -}}
  {{- $out = $out | append "" (printf "--- post: %s ---" .Permalink) "" -}}
  {{- $out = $out | append (printf "## %s" (partial "aeo-text.html" .Title)) "" -}}
  {{- $out = $out | append (printf "- URL: %s" .Permalink) -}}
  {{- with .OutputFormats.Get "markdown" -}}
    {{- $out = $out | append (printf "- Markdown: %s" .Permalink) -}}
  {{- end -}}
  {{- if not .Date.IsZero }}{{ $out = $out | append (printf "- Published: %s" (.Date.Format "2006-01-02")) }}{{ end -}}
  {{- if not .Lastmod.IsZero }}{{ $out = $out | append (printf "- Updated: %s" (.Lastmod.Format "2006-01-02")) }}{{ end -}}
  {{- $tags := partial "aeo-params-list.html" (dict "value" .Params.tags "path" (printf "tags in %s" .RelPermalink) "maps" false) -}}
  {{- with $tags }}{{ $out = $out | append (printf "- Tags: %s" (delimit (apply . "partial" "aeo-text.html" ".") ", ")) }}{{ end -}}
  {{- $out = $out | append (printf "- Words: %d" .WordCount) -}}
  {{- /* The post's own headings start at ## in the source, so its title above is
         a ## too and the document keeps one # for the site. The markdown is left
         exactly as written otherwise: rewriting a reader's markdown to fit a
         hierarchy is how a code fence stops being a code fence.

         RenderShortcodes rather than RawContent, which is the same markdown with
         the shortcodes resolved. RawContent handed the reader an unrendered
         shortcode verbatim — Hugo template syntax sitting where the content was
         supposed to be, in a file whose entire audience is machines reading it
         as prose. */ -}}
  {{- $out = $out | append "" (.RenderShortcodes | strings.TrimSpace) -}}
{{- end -}}
{{ delimit $out "\n" }}
{{- end -}}
